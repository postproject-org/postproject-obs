cmake_minimum_required(VERSION 3.21)
project(PostProjectOBS LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PostProject REQUIRED IMPORTED_TARGET postproject>=0.6.0)
add_library(registration STATIC src/registration.c)
set_target_properties(registration PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(registration PUBLIC c_std_11)
target_include_directories(registration PUBLIC src)
target_link_libraries(registration PUBLIC PkgConfig::PostProject)
target_compile_options(registration PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(BUILD_OBS_PLUGIN "Build the optional OBS frontend plugin" ON)
if(BUILD_OBS_PLUGIN)
  find_package(libobs 32 REQUIRED CONFIG)
  find_package(obs-frontend-api REQUIRED CONFIG)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  add_library(postproject-obs MODULE src/plugin.cpp src/worker.cpp)
  target_compile_features(postproject-obs PRIVATE cxx_std_17)
  target_link_libraries(postproject-obs PRIVATE registration OBS::libobs OBS::obs-frontend-api Qt6::Widgets)
  target_compile_options(postproject-obs PRIVATE -Wall -Wextra -Wpedantic -Werror)
  option(BUILD_HOST_DRIVER "Build isolated real-host acceptance automation" OFF)
  if(BUILD_HOST_DRIVER)
    add_library(postproject-obs-driver MODULE tests/driver.cpp)
    target_link_libraries(postproject-obs-driver PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets)
    target_compile_features(postproject-obs-driver PRIVATE cxx_std_17)
    target_compile_options(postproject-obs-driver PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fixture.media" "recording contract fixture")
  add_executable(registration-test tests/registration.c)
  target_link_libraries(registration-test PRIVATE registration)
  target_compile_options(registration-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME registration COMMAND registration-test
      "${CMAKE_CURRENT_BINARY_DIR}/registration.pproj"
      "${CMAKE_CURRENT_BINARY_DIR}/fixture.media")
  add_executable(readback-test tests/readback.c)
  target_link_libraries(readback-test PRIVATE registration)
  target_compile_options(readback-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  add_test(NAME capture-readback COMMAND readback-test
      "${CMAKE_CURRENT_BINARY_DIR}/readback.pproj"
      "${CMAKE_CURRENT_BINARY_DIR}/fixture.media")
endif()
